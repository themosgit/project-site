120 static bool like_match(std::string_view str, const std::string &pattern) {
121 thread_local auto regex_cache =
122 std::unordered_map<std::string, std::unique_ptr<RE2>>{};
123
124 const RE2 *re = nullptr;
125 auto it = regex_cache.find(pattern);
126 if (it != regex_cache.end()) {
127 re = it->second.get();
128 }
129
130 if (!re) {
131 std::string regex_str;
132 for (char c : pattern) {
133 if (c == '%') {
134 regex_str += ".*";
135 } else if (c == '_') {
136 regex_str += '.';
137 } else {
138 if (c == '\\' || c == '.' || c == '^' || c == '$' ||
139 c == '|' || c == '?' || c == '*' || c == '+' ||
140 c == '(' || c == ')' || c == '[' || c == ']' ||
141 c == '{' || c == '}') {
142 regex_str += '\\';
143 }
144 regex_str += c;
145 }
146 }
147
148 RE2::Options options;
149 auto new_re = std::make_unique<RE2>(regex_str, options);
150 if (!new_re->ok()) {
151 return false;
152 }
153
154 re = new_re.get();
155 regex_cache.emplace(pattern, std::move(new_re));
156 }
157
158 return RE2::FullMatch(str, *re);
159 }
