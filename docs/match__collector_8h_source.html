124 /* @brief Forward iterator over row IDs in a thread-local chunk chain. */
125 class ChainIterator {
126 IndexChunk *current_chunk;
127 uint32_t offset;
128 size_t remaining;
129
130 public:
131 ChainIterator(IndexChunk *chunk, size_t count)
132 : current_chunk(chunk), offset(0), remaining(count) {
133 if (!current_chunk)
134 remaining = 0;
135 }
136
137 inline uint32_t operator*() const { return current_chunk->ids[offset]; }
138
139 inline ChainIterator &operator++() {
140 if (remaining == 0 || !current_chunk)
141 return *this;
142 offset++;
143 remaining--;
144 if (offset >= current_chunk->count && current_chunk->next) {
145 current_chunk = current_chunk->next;
146 offset = 0;
147 }
148 return *this;
149 }
150
151 inline bool operator!=(const ChainIterator &other) const {
152 return remaining != other.remaining;
153 }

188 inline void add_match(uint32_t left, uint32_t right) {
189 if constexpr (Mode == MatchCollectionMode::BOTH) {
190 if (left_tail->count == MATCH_CHUNK_CAP) [[unlikely]] {
191 IndexChunk *new_left = alloc_chunk();
192 IndexChunk *new_right = alloc_chunk();
193 left_tail->next = new_left;
194 right_tail->next = new_right;
195 left_tail = new_left;
196 right_tail = new_right;
197 }
198 left_tail->ids[left_tail->count] = left;
199 right_tail->ids[right_tail->count] = right;
200 left_tail->count++;
201 right_tail->count++;
202 } else if constexpr (Mode == MatchCollectionMode::LEFT_ONLY) {
203 if (left_tail->count == MATCH_CHUNK_CAP) [[unlikely]] {
