73inline std::vector<SourceInfo>
74prepare_sources(const std::vector<std::tuple<size_t, DataType>> &remapped_attrs,
75 const JoinInput &build_input, const JoinInput &probe_input,
76 const PlanNode &build_node, const PlanNode &probe_node,
77 size_t build_size) {
78 std::vector<SourceInfo> sources;
79 sources.reserve(remapped_attrs.size());
80 for (const auto &[col_idx, _] : remapped_attrs) {
81 SourceInfo info;
82 info.from_build = (col_idx < build_size);
83 size_t local_idx = info.from_build ? col_idx : col_idx - build_size;
84 info.remapped_col_idx = local_idx;
85 const JoinInput &input = info.from_build ? build_input : probe_input;
86 const PlanNode &node = info.from_build ? build_node : probe_node;
87 if (input.is_columnar()) {
88 info.is_columnar = true;
89 auto *table = std::get<const ColumnarTable *>(input.data);
90 auto [actual_idx, _] = node.output_attrs[local_idx];
91 info.columnar_col = &table->columns[actual_idx];
92 } else {
93 info.is_columnar = false;
94 const auto &res = std::get<ExecuteResult>(input.data);
95 info.intermediate_col = &res[local_idx];
96 }
97 sources.push_back(info);
98 }
99 return sources;
100}

121template <MatchCollectionMode Mode>
122inline void construct_intermediate_from_buffers(
123 std::vector<ThreadLocalMatchBuffer<Mode>> &buffers,
124 const JoinInput &build_input, const JoinInput &probe_input,
125 const std::vector<std::tuple<size_t, DataType>> &remapped_attrs,
126 const PlanNode &build_node, const PlanNode &probe_node, size_t build_size,
127 ColumnarReader &columnar_reader, ExecuteResult &results) {
128
129 size_t total_matches = 0;
130 std::vector<size_t> buffer_starts(buffers.size());
131 for (size_t i = 0; i < buffers.size(); ++i) {
132 buffer_starts[i] = total_matches;
133 total_matches += buffers[i].count();
134 }
135
136 if (total_matches == 0)
137 return;
138
139 auto sources = prepare_sources(remapped_attrs, build_input, probe_input,
140 build_node, probe_node, build_size);
141
142 const size_t num_threads = THREAD_COUNT;
143 const size_t num_cols = sources.size();
144
145 using Page = mema::column_t::Page;
146 size_t total_pages_needed =
147 (total_matches + mema::CAP_PER_PAGE - 1) / mema::CAP_PER_PAGE;
148
149 for (size_t c = 0; c < num_cols; ++c) {
150 auto &col = results[c];
151 col.pages.resize(total_pages_needed);
152 col.set_row_count(total_matches);
153 }
154
155 worker_pool().execute([&](size_t t) {
156 for (size_t c = 0; c < num_cols; ++c) {
157 auto &col = results[c];
158 for (size_t p = t; p < total_pages_needed; p += num_threads) {
159 void *ptr =
160 Contest::platform::get_arena(t)
161 .alloc_chunk<Contest::platform::ChunkType::IR_PAGE>();
162 col.pages[p] = reinterpret_cast<Page *>(ptr);
163 }
164 }
165 });
166
167 worker_pool().execute([&](size_t t) {
168 if (t >= buffers.size())
169 return;
170 auto &buf = buffers[t];
171 size_t my_count = buf.count();
172 if (my_count == 0)
173 return;
174
175 size_t start = buffer_starts[t];
176 Contest::ColumnarReader::Cursor cursor;
177
178 for (size_t c = 0; c < num_cols; ++c) {
179 const auto &src = sources[c];
180 auto &dest_col = results[c];
181
182 auto left_range = buf.left_range();
183 auto right_range = buf.right_range();
184
185 if (src.is_columnar) {
186 const auto &col = *src.columnar_col;
187 if (src.from_build) {
188 size_t k = start;
189 for (uint32_t rid : left_range) {
190 dest_col.write_at(k++,
191 columnar_reader.read_value(
192 col, src.remapped_col_idx, rid,
193 col.type, cursor, true));
194 }
195 } else {
196 size_t k = start;
197 for (uint32_t rid : right_range) {
198 dest_col.write_at(k++,
199 columnar_reader.read_value(
200 col, src.remapped_col_idx, rid,
201 col.type, cursor, false));
202 }
203 }
204 } else {
205 const auto &vec = *src.intermediate_col;
206 if (src.from_build) {
207 size_t k = start;
208 for (uint32_t rid : left_range) {
209 dest_col.write_at(k++, vec[rid]);
210 }
211 } else {
212 size_t k = start;
213 for (uint32_t rid : right_range) {
214 dest_col.write_at(k++, vec[rid]);
215 }
216 }
217 }
218 }
219 });
220}
