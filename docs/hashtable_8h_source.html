113 void append(ChunkAllocator &alloc, Tuple t) {
114 if (!tail || tail->count == TUPLES_PER_CHUNK) {
115 Chunk *c = alloc.alloc();
116 if (tail)
117 tail->next = c;
118 else
119 head = c;
120 tail = c;
121 }
122 tail->data[tail->count++] = t;
123 ++total_count;
124 }

147 constexpr uint64_t k = 0x8648DBDB;

171 /* @brief Computes partition count to fit each in LLC share. */
172 size_t compute_num_partitions(size_t tuple_count, int num_threads) const {
173 size_t per_core_cache =
174 LAST_LEVEL_CACHE / Contest::platform::worker_pool().thread_count();
175 size_t target_bytes = per_core_cache / 2;
176 size_t bytes_per_tuple = sizeof(Tuple);
177 size_t tuples_per_partition = target_bytes / bytes_per_tuple;
178
179 size_t min_partitions =
180 (tuple_count + tuples_per_partition - 1) / tuples_per_partition;
181 min_partitions =
182 std::max(min_partitions, static_cast<size_t>(num_threads));
183
184 size_t p = 1;
185 while (p < min_partitions)
186 p <<= 1;
187 return std::min(p, directory.size());
188 }

197 void
198 build_partition(const std::vector<std::vector<Partition>> &thread_parts,
199 size_t p, size_t slots_per_partition, size_t base_offset,
200 size_t partition_size, int num_threads, size_t thread_id) {
201
202 const size_t slot_start = p * slots_per_partition;
203 if (partition_size == 0) {
204 for (size_t s = 0; s < slots_per_partition; ++s) {
205 directory[slot_start + s] = base_offset << 16;
206 }
207 return;
208 }
209 auto &arena = Contest::platform::get_arena(thread_id);
210 Contest::platform::ArenaVector<uint32_t> counts(arena);
211 counts.resize(slots_per_partition);
212 std::memset(counts.data(), 0, slots_per_partition * sizeof(uint32_t));
213 for (int t = 0; t < num_threads; ++t) {
214 for (Chunk *c = thread_parts[t][p].head; c; c = c->next) {
215 for (size_t i = 0; i < c->count; ++i) {
216 counts[slot_for(hash_key(c->data[i].key)) - slot_start]++;
217 }
218 }
219 }
220 Contest::platform::ArenaVector<uint32_t> offsets(arena);
221 offsets.resize(slots_per_partition);
222 uint32_t running = base_offset;
223 for (size_t s = 0; s < slots_per_partition; ++s) {
224 offsets[s] = running;
225 running += counts[s];
226 counts[s] = 0;
227 }
228 for (int t = 0; t < num_threads; ++t) {
229 for (Chunk *c = thread_parts[t][p].head; c; c = c->next) {
230 for (size_t i = 0; i < c->count; ++i) {
231 const Tuple &tup = c->data[i];
232 uint64_t h = hash_key(tup.key);
233 size_t local_slot = slot_for(h) - slot_start;
234 uint32_t idx = offsets[local_slot] + counts[local_slot]++;
235 keys_[idx] = tup.key;
236 row_ids_[idx] = tup.row_id;
237 directory[slot_start + local_slot] |= bloom_tag(h);
238 }
239 }
240 }
241 for (size_t s = 0; s < slots_per_partition; ++s) {
242 uint64_t end = offsets[s] + counts[s];
243 directory[slot_start + s] =
244 (end << 16) | (directory[slot_start + s] & 0xFFFF);
245 }
246 }

298 std::pair<uint64_t, uint64_t> find_indices(int32_t key) const noexcept {
299 if (keys_.empty())
300 return {0, 0};
301
302 uint64_t h = hash_key(key);
303 size_t slot = slot_for(h);
304 uint64_t entry = directory[slot];
305 uint16_t tag = bloom_tag(h);
306
307 if ((entry & tag) != tag)
308 return {0, 0};
309
310 uint64_t end = entry >> 16;
311 uint64_t start = (slot == 0) ? 0 : (directory[slot - 1] >> 16);
312 return {start, end};

326 void build_intermediate(const mema::column_t &column, int num_threads = 4) {
327 const size_t row_count = column.row_count();
328 if (row_count == 0)
329 return;
330
331 static constexpr size_t PARALLEL_BUILD_THRESHOLD = 1000;
332 num_threads = Contest::platform::worker_pool().thread_count();
333 if (row_count < PARALLEL_BUILD_THRESHOLD)
334 num_threads = 1;
335
336 const size_t num_slots = directory.size();
337 const size_t num_partitions =
338 compute_num_partitions(row_count, num_threads);
339 const int partition_bits = __builtin_ctzll(num_partitions);
340 const size_t slots_per_partition = num_slots / num_partitions;
341
342 std::vector<ChunkAllocator> allocators(num_threads);
343 for (int t = 0; t < num_threads; ++t)
344 allocators[t].set_arena(Contest::platform::get_arena(t));
345 std::vector<std::vector<Partition>> thread_parts(num_threads);
346 for (auto &tp : thread_parts)
347 tp.resize(num_partitions);
348
349 size_t batch = (row_count + num_threads - 1) / num_threads;
350 Contest::platform::worker_pool().execute([&, partition_bits](size_t t) {
351 size_t start = t * batch;
352 size_t end = std::min(start + batch, row_count);
353 if (start >= end)
354 return;
355 const int shift = 64 - partition_bits;
356 for (size_t i = start; i < end; ++i) {
357 int32_t val = column[i].value;
358 uint64_t h = hash_key(val);
359 size_t p = (partition_bits == 0) ? 0 : (h >> shift);
360 thread_parts[t][p].append(allocators[t],
361 {val, static_cast<uint32_t>(i)});
362 }
363 });
364
365 Contest::platform::ArenaVector<size_t> global_offsets(*arena_);
366 global_offsets.resize(num_partitions + 1);
367 std::memset(global_offsets.data(), 0,
368 (num_partitions + 1) * sizeof(size_t));
369 for (size_t p = 0; p < num_partitions; ++p) {
370 for (size_t t = 0; t < num_threads; ++t) {
371 global_offsets[p + 1] += thread_parts[t][p].total_count;
372 }
373 global_offsets[p + 1] += global_offsets[p];
374 }
375
376 size_t total = global_offsets[num_partitions];
377 if (total == 0)
378 return;
379 keys_.resize(total);
380 row_ids_.resize(total);
381
382 const int nt = num_threads;
383 Contest::platform::worker_pool().execute([&, nt](size_t t) {
384 for (size_t p = t; p < num_partitions; p += nt) {
385 build_partition(
386 thread_parts, p, slots_per_partition, global_offsets[p],
387 global_offsets[p + 1] - global_offsets[p], nt, t);
388 }
389 });
390 }

405 void build_columnar(const Column &column, int num_threads = 4) {
406 if (column.pages.empty())
407 return;
408
409 Contest::platform::ArenaVector<uint32_t> page_offsets(*arena_);
410 page_offsets.reserve(column.pages.size());

412 for (const auto &page : column.pages) {
413 page_offsets.push_back(static_cast<uint32_t>(total_rows));

415 }
416 if (total_rows == 0)

418
419 num_threads = std::clamp(
420 num_threads, 1, Contest::platform::worker_pool().thread_count());
421 if (column.pages.size() < 16)
422 num_threads = 1;

435 for (auto &tp : thread_parts)
436 tp.resize(num_partitions);
437
438 size_t batch = (num_pages + num_threads - 1) / num_threads;
439 Contest::platform::worker_pool().execute([&, partition_bits](size_t t) {
440 size_t pg_start = t * batch;
441 size_t pg_end = std::min(pg_start + batch, num_pages);
442 if (pg_start >= pg_end)
443 return;
444
445 const int shift = 64 - partition_bits;
446 for (size_t pg = pg_start; pg < pg_end; ++pg) {
447 const auto *page =
448 reinterpret_cast<const uint8_t *>(column.pages[pg]->data);
449 uint16_t n_rows = *reinterpret_cast<const uint16_t *>(page);
450 uint16_t n_vals = *reinterpret_cast<const uint16_t *>(page + 2);
451 const int32_t *vals =
452 reinterpret_cast<const int32_t *>(page + 4);
453 uint32_t base = page_offsets[pg];
454
455 if (n_rows == n_vals) {
456 for (uint16_t r = 0; r < n_rows; ++r) {
457 int32_t val = vals[r];
458 uint64_t h = hash_key(val);
459 size_t p = (partition_bits == 0) ? 0 : (h >> shift);
460 thread_parts[t][p].append(allocators[t],
461 {val, base + r});
462 }
463 } else {
464 const uint8_t *bitmap =
465 page + PAGE_SIZE - ((n_rows + 7) / 8);
466 uint16_t vi = 0;
467 for (uint16_t r = 0; r < n_rows; ++r) {
468 if (bitmap[r / 8] & (1u << (r % 8))) {
469 int32_t val = vals[vi++];
470 uint64_t h = hash_key(val);
471 size_t p = (partition_bits == 0) ? 0 : (h >> shift);
472 thread_parts[t][p].append(allocators[t],
473 {val, base + r});
474 }
475 }
476 }
477 }
478 });

481 global_offsets.resize(num_partitions + 1);

485 for (int t = 0; t < num_threads; ++t) {

488 global_offsets[p + 1] += global_offsets[p];
