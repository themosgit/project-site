116struct RegionSlice {
117 char *base = nullptr;
118 size_t capacity = 0;
119 size_t offset = 0;
120 size_t chunk_size = 0;
121
122 void init(char *b, size_t cap, size_t cs) {
123 base = b;
124 capacity = cap;
125 offset = 0;
126 chunk_size = cs;
127 }
128
135 void *allocate() noexcept {
136 if (chunk_size == 0)
137 return nullptr;
138 if (offset + chunk_size > capacity)
139 return nullptr;
140 void *ptr = base + offset;
141 offset += chunk_size;
142 return ptr;
143 }
144
152 void *allocate_n(size_t count) noexcept {
153 size_t size = chunk_size * count;
154 if (offset + size > capacity)
155 return nullptr;
156 void *ptr = base + offset;
157 offset += size;
158 return ptr;
159 }
160
169 void *alloc_bytes(size_t size, size_t align = 16) noexcept {
170 size_t aligned = (offset + align - 1) & ~(align - 1);
171 if (aligned + size > capacity)
172 return nullptr;
173 void *ptr = base + aligned;
174 offset = aligned + size;
175 return ptr;
176 }
177
178 void reset() noexcept { offset = 0; }
179 size_t used() const noexcept { return offset; }
180 size_t remaining() const noexcept { return capacity - offset; }
181};

332class ArenaManager {
333 void *memory_base_ = nullptr;
334 size_t total_size_ = 0;
335 ThreadArena thread_arenas_[SPC__THREAD_COUNT];
336 void *region_bases_[static_cast<size_t>(ChunkType::NUM_TYPES)] = {};
337
338 public:
339 explicit ArenaManager(const RegionConfig &config = DEFAULT_CONFIG) {
340 total_size_ = 0;
341 for (size_t i = 0; i < static_cast<size_t>(ChunkType::NUM_TYPES); ++i) {
342 size_t region_size = config.get(static_cast<ChunkType>(i));
343 size_t aligned = (region_size + PAGE_2MB - 1) & ~(PAGE_2MB - 1);
344 total_size_ += aligned;
345 }
346
347 memory_base_ = mmap(nullptr, total_size_, PROT_READ | PROT_WRITE,
348 MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
349 if (memory_base_ == MAP_FAILED)
350 throw std::bad_alloc();
351
352 char *ptr = static_cast<char *>(memory_base_);
353
354 for (size_t r = 0; r < static_cast<size_t>(ChunkType::NUM_TYPES); ++r) {
355 ChunkType ct = static_cast<ChunkType>(r);
356 size_t region_size =
357 (config.get(ct) + PAGE_2MB - 1) & ~(PAGE_2MB - 1);
358 size_t per_thread = region_size / SPC__THREAD_COUNT;
359 size_t chunk_size = CHUNK_SIZES[r];
360
361 region_bases_[r] = ptr;
362
363#if defined(__linux__)
364 if (REGION_PAGE_POLICY[r] == PagePolicy::HUGE_PAGES) {
365 madvise(ptr, region_size, MADV_HUGEPAGE);
366 } else {
367 madvise(ptr, region_size, MADV_NOHUGEPAGE);
368 }
369#endif
370 for (int t = 0; t < SPC__THREAD_COUNT; ++t) {
371 thread_arenas_[t].set_slice(ct, ptr + t * per_thread,
372 per_thread, chunk_size);
373 }
374
375 ptr += region_size;
376 }
377 }
378 ~ArenaManager() {
379 if (memory_base_ && memory_base_ != MAP_FAILED)
380 munmap(memory_base_, total_size_);
381 }
382
383 ArenaManager(const ArenaManager &) = delete;
384 ArenaManager &operator=(const ArenaManager &) = delete;
385
386 ThreadArena &get(size_t thread_id) noexcept {
387 return thread_arenas_[thread_id];
388 }
389
390 void reset_all() noexcept {
391 for (int t = 0; t < SPC__THREAD_COUNT; ++t)
392 thread_arenas_[t].reset();
393 }
394
395 void *region_base(ChunkType ct) const noexcept {
396 return region_bases_[static_cast<size_t>(ct)];
397 }
398
399 size_t total_size() const noexcept { return total_size_; }
400};
