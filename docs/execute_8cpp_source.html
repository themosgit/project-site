84JoinInput resolve_join_input(const Plan &plan, size_t node_idx,
85 TimingStats &stats) {
86 JoinInput input;
87 const auto &node = plan.nodes[node_idx];
88 input.node = &node;
89
90 if (const auto *scan = std::get_if<ScanNode>(&node.data)) {
91 input.data = &plan.inputs[scan->base_table_id];
92 input.table_id = scan->base_table_id;
93 } else {
94 auto result = execute_impl(plan, node_idx, false, stats);
95 input.data = std::get<ExecuteResult>(std::move(result));
96 input.table_id = 0;
97 }
98 return input;
99}

112template <MatchCollectionMode Mode>
113JoinResult execute_join_with_mode(
114 bool use_nested_loop, bool probe_is_columnar, bool is_root,
115 const UnchainedHashtable *hash_table, const JoinInput &build_input,
116 const JoinInput &probe_input, const BuildProbeConfig &config,
117 const PlanNode &build_node, const PlanNode &probe_node, JoinSetup &setup,
118 io::ColumnarReader &columnar_reader, const Plan &plan, TimingStats &stats) {
119
120 std::vector<ThreadLocalMatchBuffer<Mode>> match_buffers;
121
122 if (use_nested_loop) {
123 auto nested_loop_start = std::chrono::high_resolution_clock::now();
124 match_buffers = nested_loop_join<Mode>(
125 build_input, probe_input, config.build_attr, config.probe_attr);
126 auto nested_loop_end = std::chrono::high_resolution_clock::now();
127 stats.nested_loop_join_ms +=
128 std::chrono::duration_cast<std::chrono::milliseconds>(
129 nested_loop_end - nested_loop_start)
130 .count();
131 } else {
132 auto probe_start = std::chrono::high_resolution_clock::now();
133 if (probe_is_columnar) {
134 match_buffers = probe_columnar<Mode>(*hash_table, probe_input,
135 config.probe_attr);
136 } else {
137 const auto &probe_result =
138 std::get<ExecuteResult>(probe_input.data);
139 match_buffers = probe_intermediate<Mode>(
140 *hash_table, probe_result[config.probe_attr]);
141 }
142 auto probe_end = std::chrono::high_resolution_clock::now();
143 stats.hash_join_probe_ms +=
144 std::chrono::duration_cast<std::chrono::milliseconds>(probe_end -
145 probe_start)
146 .count();
147 }
148
149 size_t total_matches = 0;
150 for (const auto &buf : match_buffers) {
151 total_matches += buf.count();
152 }
153
154 if (is_root) {
155 auto mat_start = std::chrono::high_resolution_clock::now();
156 JoinResult final_result;
157 if (total_matches == 0) {
158 final_result = create_empty_result(config.remapped_attrs);
159 } else {
160 prepare_output_columns(
161 columnar_reader, build_input, probe_input, build_node,
162 probe_node, config.remapped_attrs, build_input.output_size());
163
164 final_result = materialize_from_buffers<Mode>(
165 match_buffers, build_input, probe_input, config.remapped_attrs,
166 build_node, probe_node, build_input.output_size(),
167 columnar_reader, plan);
168 }
169 auto mat_end = std::chrono::high_resolution_clock::now();
170 stats.materialize_ms +=
171 std::chrono::duration_cast<std::chrono::milliseconds>(mat_end -
172 mat_start)
173 .count();
174 return final_result;
175 } else {
176 auto inter_start = std::chrono::high_resolution_clock::now();
177 if (total_matches > 0) {
178 prepare_output_columns(
179 columnar_reader, build_input, probe_input, build_node,
180 probe_node, config.remapped_attrs, build_input.output_size());
181
182 construct_intermediate_from_buffers<Mode>(
183 match_buffers, build_input, probe_input, config.remapped_attrs,
184 build_node, probe_node, build_input.output_size(),
185 columnar_reader, setup.results);
186 }
187 auto inter_end = std::chrono::high_resolution_clock::now();
188 stats.intermediate_ms +=
189 std::chrono::duration_cast<std::chrono::milliseconds>(inter_end -
190 inter_start)
191 .count();
192 return std::move(setup.results);
193 }

207JoinResult execute_impl(const Plan &plan, size_t node_idx, bool is_root,
208 TimingStats &stats) {
209 auto &node = plan.nodes[node_idx];
210
211 if (!std::holds_alternative<JoinNode>(node.data)) {
212 return ExecuteResult{};
213 }
214
215 const auto &join = std::get<JoinNode>(node.data);
216 const auto &output_attrs = node.output_attrs;
217 const auto &left_node = plan.nodes[join.left];
218 const auto &right_node = plan.nodes[join.right];
219
220 JoinInput left_input = resolve_join_input(plan, join.left, stats);
221 JoinInput right_input = resolve_join_input(plan, join.right, stats);
222
223 /* Build/probe selection smaller input = build side; remaps output_attrs. */
224 auto setup_start = std::chrono::high_resolution_clock::now();
225 auto config =
226 select_build_probe_side(join, left_input, right_input, output_attrs);
227 const JoinInput &build_input = config.build_left ? left_input : right_input;
228 const JoinInput &probe_input = config.build_left ? right_input : left_input;
229 const auto &build_node = config.build_left ? left_node : right_node;
230 const auto &probe_node = config.build_left ? right_node : left_node;
231
232 bool build_is_columnar = build_input.is_columnar();
233 bool probe_is_columnar = probe_input.is_columnar();
234
235 /* Nested loop for <8 rows (L1-resident, no hash overhead, SIMD). */
236 const size_t HASH_TABLE_THRESHOLD = 8;
237 size_t build_rows = build_input.row_count(config.build_attr);
238 bool use_nested_loop = (build_rows < HASH_TABLE_THRESHOLD);
239
240 /* Pre-allocate ExecuteResult; ColumnarReader PageIndex built lazily. */
241 JoinSetup setup = setup_join(build_input, probe_input, build_node,
242 probe_node, left_node, right_node, left_input,
243 right_input, output_attrs, build_rows);
244 auto setup_end = std::chrono::high_resolution_clock::now();
245 auto setup_elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
246 setup_end - setup_start);
247 stats.setup_ms += setup_elapsed.count();
248
249 /* Skip unused-side row IDs if output needs only one side. */
250 MatchCollectionMode collection_mode = determine_collection_mode(
251 config.remapped_attrs, config.build_left ? left_input.output_size()
252 : right_input.output_size());
253
254 /* Build hash table if needed (before mode dispatch). */
255 std::optional<UnchainedHashtable> hash_table;
256 if (!use_nested_loop) {
257 auto build_start = std::chrono::high_resolution_clock::now();
258 hash_table =
259 build_is_columnar
260 ? build_from_columnar(build_input, config.build_attr)
261 : build_from_intermediate(build_input, config.build_attr);
262 auto build_end = std::chrono::high_resolution_clock::now();
263 stats.hashtable_build_ms +=
264 std::chrono::duration_cast<std::chrono::milliseconds>(build_end -
265 build_start)
266 .count();
267 }
268
276 switch (collection_mode) {
277 case MatchCollectionMode::BOTH:
278 return execute_join_with_mode<MatchCollectionMode::BOTH>(
279 use_nested_loop, probe_is_columnar, is_root,
280 use_nested_loop ? nullptr : &(*hash_table), build_input,
281 probe_input, config, build_node, probe_node, setup,
282 setup.columnar_reader, plan, stats);
283
284 case MatchCollectionMode::LEFT_ONLY:
285 return execute_join_with_mode<MatchCollectionMode::LEFT_ONLY>(
286 use_nested_loop, probe_is_columnar, is_root,
287 use_nested_loop ? nullptr : &(*hash_table), build_input,
288 probe_input, config, build_node, probe_node, setup,
289 setup.columnar_reader, plan, stats);
290
291 case MatchCollectionMode::RIGHT_ONLY:
292 return execute_join_with_mode<MatchCollectionMode::RIGHT_ONLY>(
293 use_nested_loop, probe_is_columnar, is_root,
294 use_nested_loop ? nullptr : &(*hash_table), build_input,
295 probe_input, config, build_node, probe_node, setup,
296 setup.columnar_reader, plan, stats);
297 }
298 /* unreachable */
299 return ExecuteResult{};
300}

311ColumnarTable execute(const Plan &plan, void *context, TimingStats *stats_out) {
312
313 Contest::platform::g_arena_manager.reset_all();
314 auto total_start = std::chrono::high_resolution_clock::now();
315 TimingStats stats;
316 auto result = execute_impl(plan, plan.root, true, stats);
317 auto total_end = std::chrono::high_resolution_clock::now();
318 auto total_elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
319 total_end - total_start);
320 stats.total_execution_ms = total_elapsed.count();
321 if (stats_out) {
322 *stats_out = stats;
323 }
324 return std::move(std::get<ColumnarTable>(result));
325}
