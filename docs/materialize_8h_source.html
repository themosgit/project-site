85template <MatchCollectionMode Mode, typename BuilderType, typename ReaderFunc,
86 typename InitBuilderFunc>
87inline void materialize_column_from_buffers(
88 Column &dest_col, std::vector<ThreadLocalMatchBuffer<Mode>> &buffers,
89 size_t total_matches, ReaderFunc &&read_value,
90 InitBuilderFunc &&init_builder, bool from_build, size_t est_bytes_per_row) {
91
92 if (total_matches == 0)
93 return;
94
95 const int num_threads = THREAD_COUNT;
96
97 size_t matches_per_thread = (total_matches + num_threads - 1) / num_threads;
98 size_t usable_per_page = PAGE_SIZE - 256;
99 size_t rows_per_page = std::max(1ul, usable_per_page / est_bytes_per_row);
100 size_t pages_per_thread =
101 (matches_per_thread + rows_per_page - 1) / rows_per_page + 10;
102
103 std::vector<Column> thread_columns;
104 thread_columns.reserve(num_threads);
105 for (int i = 0; i < num_threads; ++i) {
106 thread_columns.emplace_back(dest_col.type);
107 }
108
109 std::vector<char *> thread_page_memory(num_threads, nullptr);
110
111 worker_pool().execute([&](size_t t) {
112 if (t >= buffers.size())
113 return;
114 auto &buf = buffers[t];
115 size_t my_count = buf.count();
116 if (my_count == 0)
117 return;
118
119 auto &arena = platform::get_arena(t);
120 size_t alloc_size = pages_per_thread * PAGE_SIZE;
121 thread_page_memory[t] =
122 static_cast<char *>(arena.alloc(alloc_size, PAGE_SIZE));
123
124 Column &local_col = thread_columns[t];
125 size_t used_pages = 0;
126
127 ColumnarReader::Cursor cursor;
128
129 auto page_allocator = [&]() -> Page * {
130 Page *p;
131 if (used_pages < pages_per_thread) {
132 p = reinterpret_cast<Page *>(thread_page_memory[t] +
133 used_pages * PAGE_SIZE);
134 used_pages++;
135 } else {
136 p = static_cast<Page *>(arena.alloc(PAGE_SIZE, PAGE_SIZE));
137 }
138 local_col.pages.push_back(p);
139 return p;
140 };
141
142 BuilderType builder = init_builder(page_allocator);
143 builder.prepare(my_count);
144
145 const size_t check_interval = BuilderType::MIN_ROWS_PER_PAGE_CHECK;
146 size_t rows_since_check = 0;
147
148 auto range = from_build ? buf.left_range() : buf.right_range();
149
150 for (uint32_t row_id : range) {
151 bool flushed = builder.add(read_value(row_id, cursor));
152
153 if (flushed) {
154 rows_since_check = 0;
155 } else {
156 rows_since_check++;
157 if (rows_since_check >= check_interval) {
158 if (builder.should_check_overflow()) {
159 builder.save_to_page(builder.current_page);
160 rows_since_check = 0;
161 }
162 if (rows_since_check > check_interval * 2)
163 rows_since_check = 0;
164 }
165 }
166 }
167
168 if (builder.num_rows != 0) {
169 builder.save_to_page(builder.current_page);
170 }
171 });
172
173 for (auto &thread_col : thread_columns) {
174 for (auto *page : thread_col.pages) {
175 dest_col.pages.push_back(page);
176 }
177 thread_col.pages.clear();
178 }
185 dest_col.assign_mapped_memory(get_arena_mapped_memory());
186}

202template <MatchCollectionMode Mode>
203inline void materialize_single_column_from_buffers(
204 Column &dest_col, size_t col_idx, size_t build_size,
205 std::vector<ThreadLocalMatchBuffer<Mode>> &buffers, size_t total_matches,
206 const JoinInput &build_input, const JoinInput &probe_input,
207 const PlanNode &build_node, const PlanNode &probe_node,
208 ColumnarReader &columnar_reader, const Plan &plan) {
209
210 auto [input, node, local_idx] = resolve_input_source(
211 col_idx, build_size, build_input, build_node, probe_input, probe_node);
212 bool from_build = col_idx < build_size;
213
214 const Column *col_source = nullptr;
215 const mema::column_t *inter_source = nullptr;
216
217 if (input.is_columnar()) {
218 auto *table = std::get<const ColumnarTable *>(input.data);
219 auto [actual_idx, _] = node.output_attrs[local_idx];
220 col_source = &table->columns[actual_idx];
221 } else {
222 const auto &res = std::get<ExecuteResult>(input.data);
223 inter_source = &res[local_idx];
224 }
225
226 auto reader = [&](uint32_t rid, ColumnarReader::Cursor &cursor,
227 DataType type) {
228 if (col_source) {
229 return columnar_reader.read_value(*col_source, local_idx, rid, type,
230 cursor, from_build);
231 }
232 return (*inter_source)[rid];
233 };
234
235 if (dest_col.type == DataType::INT32) {
236 auto init = [](std::function<Page *()> alloc) {
237 return Int32PageBuilder(std::move(alloc));
238 };
239 materialize_column_from_buffers<Mode, Int32PageBuilder>(
240 dest_col, buffers, total_matches,
241 [&](uint32_t rid, ColumnarReader::Cursor &cursor) {
242 return reader(rid, cursor, DataType::INT32);
243 },
244 init, from_build, 4);
245 return;
246 }
247
248 const Column *str_src_ptr = col_source;
249 if (!str_src_ptr && inter_source) {
250 str_src_ptr = &plan.inputs[inter_source->source_table]
251 .columns[inter_source->source_column];
252 }
253
254 auto init = [str_src_ptr](std::function<Page *()> alloc) {
255 return VarcharPageBuilder(*str_src_ptr, std::move(alloc));
256 };
257
258 materialize_column_from_buffers<Mode, VarcharPageBuilder>(
259 dest_col, buffers, total_matches,
260 [&](uint32_t rid, ColumnarReader::Cursor &cursor) {
261 return reader(rid, cursor, DataType::VARCHAR);
262 },
263 init, from_build, 35);
264}

289template <MatchCollectionMode Mode>
290inline ColumnarTable materialize_from_buffers(
291 std::vector<ThreadLocalMatchBuffer<Mode>> &buffers,
292 const JoinInput &build_input, const JoinInput &probe_input,
293 const std::vector<std::tuple<size_t, DataType>> &remapped_attrs,
294 const PlanNode &build_node, const PlanNode &probe_node, size_t build_size,
295 ColumnarReader &columnar_reader, const Plan &plan) {
296
297 size_t total_matches = 0;
298 for (const auto &buf : buffers) {
299 total_matches += buf.count();
300 }
301
302 ColumnarTable result;
303 result.num_rows = total_matches;
304
305 if (total_matches == 0) {
306 for (auto [_, dtype] : remapped_attrs) {
307 result.columns.emplace_back(dtype);
308 }
309 return result;
310 }
311
312 for (size_t out_idx = 0; out_idx < remapped_attrs.size(); ++out_idx) {
313 auto [col_idx, data_type] = remapped_attrs[out_idx];
314 result.columns.emplace_back(data_type);
315 Column &dest_col = result.columns.back();
316 materialize_single_column_from_buffers<Mode>(
317 dest_col, col_idx, build_size, buffers, total_matches, build_input,
318 probe_input, build_node, probe_node, columnar_reader, plan);
319 }
320 return result;
321}
