25/* @brief Gets string data from a VARCHAR column page. */
26inline std::pair<const char *, uint16_t>
27get_string_view(const Column &src_col, int32_t page_idx, int32_t offset_idx) {
28 auto *page = reinterpret_cast<uint8_t *>(src_col.pages[page_idx]->data);
29 auto num_valid = *reinterpret_cast<uint16_t *>(page + 2);
30 auto *offset_array = reinterpret_cast<uint16_t *>(page + 4);
31 char *char_begin = reinterpret_cast<char *>(page + 4 + num_valid * 2);
32
33 uint16_t end_off = offset_array[offset_idx];
34 uint16_t start_off = (offset_idx == 0) ? 0 : offset_array[offset_idx - 1];
35
36 return {char_begin + start_off, static_cast<uint16_t>(end_off - start_off)};
37}

45struct alignas(8) BitmapAccumulator {
46 std::vector<uint8_t> buffer;
47 uint8_t pending_bits = 0;
48 int bit_count = 0;
49
50 void reserve(size_t count) {
51 buffer.clear();
52 buffer.reserve((count + 7) / 8);
53 }
54
55 inline void add_bit(bool set) {
56 if (set)
57 pending_bits |= (1u << bit_count);
58 if (++bit_count == 8) {
59 buffer.push_back(pending_bits);
60 pending_bits = 0;
61 bit_count = 0;
62 }
63 }
64
65 void flush_to_memory(uint8_t *dest_ptr) {
66 if (bit_count > 0) {
67 buffer.push_back(pending_bits);
68 pending_bits = 0;
69 bit_count = 0;
70 }
71 if (!buffer.empty()) {
72 std::memcpy(dest_ptr, buffer.data(), buffer.size());
73 }
74 buffer.clear();
75 }
76
77 size_t current_byte_size() const {
78 return buffer.size() + (bit_count > 0 ? 1 : 0);
79 }
80};

204 bitmap.reserve(chunk_matches);

208 if (val.is_null()) {
209 bitmap.add_bit(false);
210 num_rows++;
211 return false;
212 }
213
214 int32_t page_idx, offset_idx;
215 mema::value_t::decode_string(val.value, page_idx, offset_idx);
216
217 if (offset_idx == mema::value_t::LONG_STRING_OFFSET) {

222 }
223

226
227 if (str_len > PAGE_SIZE - 512) {
228 if (num_rows > 0)
229 flush_current_page();

231 return true;
232 }
233

236

239 size_t physical_space = PAGE_SIZE - ptr_offset - 64;
240
241 size_t needed = 4 + (offsets.size() + 1) * 2 + current_char_bytes +
242 str_len + bitmap.current_byte_size() + 10;
243
244 bool flushed = false;
245 if (num_rows == 65535 || needed > PAGE_SIZE ||
246 (offsets.size() * 2 >= current_gap_size) ||
247 str_len > physical_space) {
248 flush_current_page();
249 init_new_page();

252 size_t reduced_gap = 256;
253 string_write_ptr = reinterpret_cast<char *>(current_page->data +
254 4 + reduced_gap);
255 current_gap_size = reduced_gap;
256 }
257 }
258
259 std::memcpy(string_write_ptr, str_ptr, str_len);
260 string_write_ptr += str_len;
261 current_char_bytes += str_len;
262
263 offsets.push_back(static_cast<uint16_t>(current_char_bytes));
264 bitmap.add_bit(true);
265 num_rows++;
266
267 return flushed;
268 }

314 /* @brief Finalize page: write header, compact strings, write bitmap. */
315 void finalize_page() {
316 uint8_t *page_base = reinterpret_cast<uint8_t *>(current_page->data);
317 size_t offsets_size = offsets.size() * 2;
318 char *chars_start_actual =
319 reinterpret_cast<char *>(page_base + 4 + offsets_size);
320 char *chars_gap_end = string_write_ptr;
321
322 *reinterpret_cast<uint16_t *>(page_base) = num_rows;
323 *reinterpret_cast<uint16_t *>(page_base + 2) =
324 static_cast<uint16_t>(offsets.size());
325
326 std::memcpy(page_base + 4, offsets.data(), offsets_size);
327
328 if (current_char_bytes > 0) {
329 std::memmove(chars_start_actual, chars_gap_end - current_char_bytes,
330 current_char_bytes);
331 }
332
333 size_t bmp_size = bitmap.current_byte_size();
334 bitmap.flush_to_memory(page_base + PAGE_SIZE - bmp_size);
335 }

345 void copy_long_string_pages(int32_t start_page_idx) {
346 int32_t curr_idx = start_page_idx;
347 while (true) {
348 auto *src = src_col.pages[curr_idx]->data;
349 auto *dest = alloc_page()->data;
350 std::memcpy(dest, src, PAGE_SIZE);
351 if (++curr_idx >= static_cast<int32_t>(src_col.pages.size()))
352 break;
353 if (*reinterpret_cast<uint16_t *>(src_col.pages[curr_idx]->data) !=
354 0xfffe)
355 break;
356 }
357 num_rows = 0;
358 offsets.clear();
359 bitmap.buffer.clear();
360 current_char_bytes = 0;
361 current_page = nullptr;
362 }

372 void save_long_string_buffer(const char *data_ptr, size_t total_len) {
373 size_t offset = 0;
374 bool first_page = true;
375 while (offset < total_len) {
376 auto *page = alloc_page()->data;
377 *reinterpret_cast<uint16_t *>(page) = first_page ? 0xffff : 0xfffe;
378 first_page = false;
379 size_t len = std::min(total_len - offset, PAGE_SIZE - 4);
380 *reinterpret_cast<uint16_t *>(page + 2) =
381 static_cast<uint16_t>(len);
382 std::memcpy(page + 4, data_ptr + offset, len);
383 offset += len;
384 }
385 num_rows = 0;
386 offsets.clear();
387 bitmap.buffer.clear();
388 current_char_bytes = 0;
389 current_page = nullptr;
390 }
