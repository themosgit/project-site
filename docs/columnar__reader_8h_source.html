120 inline void build(const Column &column) {
121 uint32_t total = 0;
122 cumulative_rows.reserve(column.pages.size());
123 page_prefix_sums.reserve(column.pages.size());
124 all_dense = true;
125
126 /* Dense INT32 detection: track if all pages are dense and uniform */
127 bool candidate_dense_int32 = (column.type == DataType::INT32);
128 uint16_t first_full_page_rows = 0;
129
130 size_t page_idx = 0;
131 for (auto *page_obj : column.pages) {
132 auto *page = page_obj->data;
133 auto num_rows = *reinterpret_cast<const uint16_t *>(page);
134
135 if (SPC_UNLIKELY(num_rows == 0xfffe)) {
136 all_dense = false;
137 candidate_dense_int32 = false;
138 } else if (SPC_UNLIKELY(num_rows == 0xffff)) {
139 all_dense = false;
140 candidate_dense_int32 = false;
141 total += 1;
142 } else {
143 total += num_rows;
144 }
145 cumulative_rows.push_back(total);
146
147 auto num_values = *reinterpret_cast<const uint16_t *>(page + 2);
148 platform::ArenaVector<uint32_t> prefix_sums(*arena_);
149
150 /* sparse page: build prefix sums for bitmap popcount */
151 if (num_rows != 0xfffe && num_rows != 0xffff &&
152 num_rows != num_values) {
153 all_dense = false;
154 candidate_dense_int32 = false;
155
156 size_t bitmap_size = (num_rows + 7) / 8;
157 auto *bitmap_bytes = reinterpret_cast<const uint8_t *>(
158 page + PAGE_SIZE - bitmap_size);
159
160 size_t num_chunks = (num_rows + 63) / 64;
161 prefix_sums.reserve(num_chunks);
162 uint32_t sum = 0;
163 for (size_t i = 0; i < num_chunks; ++i) {
164 prefix_sums.push_back(sum);
165 uint64_t word = 0;
166 size_t offset = i * 8;
167 size_t remaining =
168 bitmap_size > offset ? bitmap_size - offset : 0;
169 size_t bytes_to_read = std::min(remaining, size_t(8));
170 if (bytes_to_read > 0) {
171 std::memcpy(&word, bitmap_bytes + offset,
172 bytes_to_read);
173 }
174 sum += __builtin_popcountll(word);
175 }
176 }
177 page_prefix_sums.push_back(std::move(prefix_sums));
178
179 /* Track rows_per_full_page: use first page as reference */
180 if (candidate_dense_int32) {
181 if (page_idx == 0) {
182 first_full_page_rows = num_rows;
183 } else if (page_idx < column.pages.size() - 1) {
184 /* All full pages (except last) must have same row count */
185 if (num_rows != first_full_page_rows) {
186 candidate_dense_int32 = false;
187 }
188 }
189 }
190
191 page_idx++;
192 }
193
194 /* Finalize dense INT32 metadata */
195 is_dense_int32 = candidate_dense_int32 && !column.pages.empty() &&
196 first_full_page_rows > 0;
197 if (is_dense_int32) {
198 rows_per_full_page = first_full_page_rows;
199 pages_ptr = &column.pages;
200 }
201 }
202
203 /* @brief Finds the page containing row_id via binary search. */
204 inline size_t find_page(uint32_t row_id) const {
205 auto it = std::upper_bound(cumulative_rows.begin(),
206 cumulative_rows.end(), row_id);
207 return static_cast<size_t>(it - cumulative_rows.begin());
208 }
209
210 /* @brief Returns the starting row number for a given page. */
211 inline uint32_t page_start_row(size_t page_num) const {
212 return page_num == 0 ? 0 : cumulative_rows[page_num - 1];
213 }

264 auto &arena = platform::get_arena(0);
265 auto &indices = page_indices_[static_cast<size_t>(side)];

300 inline mema::value_t read_value(const Column &column, size_t col_idx,
301 uint32_t row_id, DataType data_type,
302 Cursor &cursor, Side side) const {
303 const auto &page_index =
304 page_indices_[static_cast<size_t>(side)][col_idx];
305
306 /* Dense INT32 fast path: O(1) arithmetic lookup, bypasses cursor */
307 if (data_type == DataType::INT32) {
308 if (SPC_LIKELY(page_index.is_dense_int32)) {
309 return mema::value_t{read_dense_int32(page_index, row_id)};
310 }
311 }
312
313 uint64_t current_version =
314 global_version.load(std::memory_order_relaxed);
315
316 /* Cursor hit: same version, same side, same column, row in range */
317 if (SPC_LIKELY(cursor.version == current_version &&
318 cursor.cached_side == side &&
319 col_idx == cursor.cached_col &&
320 row_id >= cursor.cached_start &&
321 row_id < cursor.cached_end)) {
322 uint32_t local_row = row_id - cursor.cached_start;
323 if (SPC_LIKELY(cursor.is_dense)) {
324 if (data_type == DataType::INT32) {
325 return mema::value_t{cursor.data_ptr[local_row]};
326 } else {
327 return mema::value_t::encode_string(
328 cursor.page_idx_val, static_cast<int32_t>(local_row));
329 }
330 }
331 if (SPC_UNLIKELY(cursor.is_special)) {
332 return mema::value_t::encode_string(
333 cursor.page_idx_val, mema::value_t::LONG_STRING_OFFSET);
334 }
335
336 return read_sparse(local_row, data_type, cursor);
337 }
338
339 /* Sequential access optimization: skip binary search for next page */
340 if (SPC_LIKELY(cursor.version == current_version &&
341 cursor.cached_side == side &&
342 col_idx == cursor.cached_col &&
343 row_id == cursor.cached_end)) {
344 size_t next_page = cursor.cached_page + 1;
345 if (SPC_LIKELY(next_page < page_index.cumulative_rows.size())) {
346 load_page_into_cursor(column, page_index, next_page, col_idx,
347 current_version, side, cursor);
348 return read_value(column, col_idx, row_id, data_type, cursor,
349 side);
350 }
351 }
352
353 /* Slow path: O(log P) binary search */
354 return read_value_slow(column, col_idx, row_id, data_type, cursor, side,
355 current_version);
356 }

401 inline void load_page_into_cursor(const Column &column,
402 const PageIndex &page_index,
403 size_t page_num, size_t col_idx,
404 uint64_t current_version, Side side,
405 Cursor &cursor) const {
406 cursor.version = current_version;
407 cursor.cached_col = col_idx;
408 cursor.cached_page = page_num;
409 cursor.cached_start = page_index.page_start_row(page_num);
410 cursor.cached_end = page_index.cumulative_rows[page_num];
411 cursor.page_idx_val = static_cast<int32_t>(page_num);
412 cursor.col_all_dense = page_index.all_dense;
413 cursor.cached_side = side;
414
415 auto *page_data = column.pages[page_num]->data;
416 auto num_rows = *reinterpret_cast<const uint16_t *>(page_data);
417 auto num_values = *reinterpret_cast<const uint16_t *>(page_data + 2);
418
419 cursor.is_special = (num_rows == 0xffff);
420 cursor.is_dense = (num_rows == num_values);
421 cursor.data_ptr = reinterpret_cast<const int32_t *>(page_data + 4);
422
423 if (!cursor.is_dense && !cursor.is_special) {
424 size_t bitmap_size = (num_rows + 7) / 8;
425 cursor.bitmap_ptr = reinterpret_cast<const uint8_t *>(
426 page_data + PAGE_SIZE - bitmap_size);
427 cursor.prefix_sum_ptr =
428 page_index.page_prefix_sums[page_num].data();
429 }

451 /* @brief Reads from sparse pages using bitmap and popcount. */
452 inline mema::value_t read_sparse(uint32_t local_row, DataType data_type,
453 const Cursor &cursor) const {
454 bool is_valid =
455 cursor.bitmap_ptr[local_row >> 3] & (1u << (local_row & 7));
456
457 if (!is_valid)
458 return mema::value_t{mema::value_t::NULL_VALUE};
459
460 size_t chunk_idx = local_row >> 6;
461 size_t bit_offset = local_row & 0x3F;
462 uint16_t data_idx = cursor.prefix_sum_ptr[chunk_idx];
463
464 uint64_t word;
465 std::memcpy(&word, cursor.bitmap_ptr + (chunk_idx * 8), 8);
466 data_idx += __builtin_popcountll(word & ((1ULL << bit_offset) - 1));
467
468 if (data_type == DataType::INT32) {
469 return mema::value_t{cursor.data_ptr[data_idx]};
470 } else {
471 return mema::value_t::encode_string(cursor.page_idx_val,
472 static_cast<int32_t>(data_idx));
473 }
474 }
