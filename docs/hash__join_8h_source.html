87template <MatchCollectionMode Mode>
88inline std::vector<ThreadLocalMatchBuffer<Mode>>
89probe_intermediate(const UnchainedHashtable &hash_table,
90 const mema::column_t &probe_column) {
91 const auto *keys = hash_table.keys();
92 const auto *row_ids = hash_table.row_ids();
93
94 size_t pool_size = THREAD_COUNT;
95 std::vector<ThreadLocalMatchBuffer<Mode>> local_buffers(pool_size);
96
97 const size_t num_pages = probe_column.pages.size();
98 const size_t probe_count = probe_column.row_count();
99 std::atomic<size_t> page_counter(0);
100
101 worker_pool().execute([&](size_t thread_id) {
102 local_buffers[thread_id] = ThreadLocalMatchBuffer<Mode>(
103 Contest::platform::get_arena(thread_id));
104 auto &local_buf = local_buffers[thread_id];
105
106 while (true) {
107 size_t page_idx = page_counter.fetch_add(1);
108 if (page_idx >= num_pages)
109 break;
110
111 size_t base_row = page_idx * mema::CAP_PER_PAGE;
112 size_t page_end =
113 std::min(base_row + mema::CAP_PER_PAGE, probe_count);
114
115 constexpr size_t PREFETCH_DIST = 8;
116 for (size_t idx = base_row; idx < page_end; ++idx) {
117 if (idx + PREFETCH_DIST < page_end) {
118 const mema::value_t &future_key =
119 probe_column[idx + PREFETCH_DIST];
120 if (!future_key.is_null()) {
121 hash_table.prefetch_slot(future_key.value);
122 }
123 }
124
125 const mema::value_t &key = probe_column[idx];
126
127 if (!key.is_null()) {
128 int32_t key_val = key.value;
129 auto [start_idx, end_idx] =
130 hash_table.find_indices(key_val);
131
132 for (uint64_t i = start_idx; i < end_idx; ++i) {
133 if (keys[i] == key_val) {
134 local_buf.add_match(row_ids[i],
135 static_cast<uint32_t>(idx));
136 }
137 }
138 }
139 }
140 }
141 });
142
143 return local_buffers;
144}

162 const auto *row_ids = hash_table.row_ids();
163
164 auto *table = std::get<const ColumnarTable *>(probe_input.data);
165 auto [actual_idx_col, _] = probe_input.node->output_attrs[probe_attr];
166 const Column &probe_col = table->columns[actual_idx_col];
167 size_t num_pages = probe_col.pages.size();
168
169 auto &arena = Contest::platform::get_arena(0);
170 Contest::platform::ArenaVector<uint32_t> page_offsets(arena);
171 page_offsets.reserve(num_pages);
172 uint32_t running_offset = 0;
173
174 for (const auto *page_obj : probe_col.pages) {
175 page_offsets.push_back(running_offset);
176 auto num_rows = *reinterpret_cast<const uint16_t *>(page_obj->data);
177 running_offset += num_rows;
178 }
179
180 size_t pool_size = THREAD_COUNT;
181 std::vector<ThreadLocalMatchBuffer<Mode>> local_buffers(pool_size);
182
183 std::atomic<size_t> page_counter(0);
184 worker_pool().execute([&](size_t thread_id) {
185 local_buffers[thread_id] = ThreadLocalMatchBuffer<Mode>(
186 Contest::platform::get_arena(thread_id));
187 auto &local_buf = local_buffers[thread_id];
188
189 while (true) {
190 size_t page_idx = page_counter.fetch_add(1);
191 if (page_idx >= num_pages)
192 break;
193
194 auto *page = probe_col.pages[page_idx]->data;
195 auto num_rows = *reinterpret_cast<const uint16_t *>(page);
196 auto num_values = *reinterpret_cast<const uint16_t *>(page + 2);
197 auto *data_begin = reinterpret_cast<const int32_t *>(page + 4);
198 uint32_t probe_row_id = page_offsets[page_idx];
199
200 if (num_rows == num_values) {
201 constexpr int PREFETCH_DIST = 32;
202 for (uint16_t i = 0; i < num_rows; ++i) {
203 hash_table.prefetch_slot(data_begin[i + PREFETCH_DIST]);
204
205 int32_t key_val = data_begin[i];
206 auto [start_idx, end_idx] =
207 hash_table.find_indices(key_val);
208
209 for (uint64_t j = start_idx; j < end_idx; ++j) {
210 if (keys[j] == key_val) {
211 local_buf.add_match(row_ids[j], probe_row_id);
212 }
213 }
214 probe_row_id++;
215 }
216 } else {
217 auto *bitmap = reinterpret_cast<const uint8_t *>(
218 page + PAGE_SIZE - (num_rows + 7) / 8);
219 constexpr int PREFETCH_DIST = 32;
220 uint16_t data_idx = 0;
221 for (uint16_t i = 0; i < num_rows; ++i) {
222 bool is_valid = bitmap[i / 8] & (1u << (i % 8));
223 if (is_valid) {
224 hash_table.prefetch_slot(
225 data_begin[data_idx + PREFETCH_DIST]);
226
227 int32_t key_val = data_begin[data_idx++];
228 auto [start_idx, end_idx] =
229 hash_table.find_indices(key_val);
230
231 for (uint64_t j = start_idx; j < end_idx; ++j) {
232 if (keys[j] == key_val) {
233 local_buf.add_match(row_ids[j], probe_row_id);
234 }
235 }
236 }

238 }
