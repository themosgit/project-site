108inline BuildProbeConfig select_build_probe_side(
109 const JoinNode &join, const JoinInput &left_input,
110 const JoinInput &right_input,
111 const std::vector<std::tuple<size_t, DataType>> &output_attrs) {
112 BuildProbeConfig config;
113
114 size_t left_rows = left_input.row_count(join.left_attr);
115 size_t right_rows = right_input.row_count(join.right_attr);
116 config.build_left = left_rows <= right_rows;
117
118 config.build_attr = config.build_left ? join.left_attr : join.right_attr;
119 config.probe_attr = config.build_left ? join.right_attr : join.left_attr;
120
121 config.remapped_attrs = output_attrs;
122 size_t left_size = left_input.output_size();
123 size_t build_size =
124 config.build_left ? left_size : right_input.output_size();
125
126 if (!config.build_left) {
127 for (auto &[col_idx, dtype] : config.remapped_attrs) {
128 if (col_idx < left_size) {
129 col_idx = build_size + col_idx;
130 } else {
131 col_idx = col_idx - left_size;
132 }
133 }
134 }
135 return config;
136}

146inline MatchCollectionMode determine_collection_mode(
147 const std::vector<std::tuple<size_t, DataType>> &remapped_attrs,
148 size_t build_size) {
149
150 bool needs_build = false;
151 bool needs_probe = false;
152
153 for (const auto &[col_idx, dtype] : remapped_attrs) {
154 if (col_idx < build_size) {
155 needs_build = true;
156 } else {
157 needs_probe = true;
158 }
159
160 if (needs_build && needs_probe) {
161 return MatchCollectionMode::BOTH;
162 }
163 }
164
165 if (needs_build && !needs_probe) {
166 return MatchCollectionMode::LEFT_ONLY;
167 }
168 if (needs_probe && !needs_build) {
169 return MatchCollectionMode::RIGHT_ONLY;
170 }
171
172 return MatchCollectionMode::BOTH;
173}

180inline ExecuteResult initialize_output_columns(
181 const std::vector<std::tuple<size_t, DataType>> &output_attrs,
182 const PlanNode &left_node, const PlanNode &right_node,
183 const JoinInput &left_input, const JoinInput &right_input,
184 size_t estimated_rows) {
185 ExecuteResult results;
186 results.reserve(output_attrs.size());
187 size_t left_size = left_input.output_size();
188
189 auto set_column_metadata = [](mema::column_t &col, const JoinInput &input,
190 const PlanNode &node, size_t col_idx) {
191 auto [actual_col_idx, _] = node.output_attrs[col_idx];
192 if (input.is_columnar()) {
193 col.source_table = input.table_id;
194 col.source_column = actual_col_idx;
195 } else {
196 const auto &result = std::get<ExecuteResult>(input.data);
197 col.source_table = result[col_idx].source_table;
198 col.source_column = result[col_idx].source_column;
199 }
200 };
201
202 for (size_t i = 0; i < output_attrs.size(); ++i) {
203 auto [col_idx, _] = output_attrs[i];
204 auto [input, node, local_idx] = resolve_input_source(
205 col_idx, left_size, left_input, left_node, right_input, right_node);
206
207 mema::column_t col;
208 set_column_metadata(col, input, node, local_idx);
209 results.push_back(std::move(col));
210 }
211
212 return results;
213}

263inline platform::ArenaVector<const Column *>
264collect_needed_columns(const JoinInput &input, const PlanNode &node,
265 const platform::ArenaVector<uint8_t> &needed,
266 platform::ThreadArena &arena) {
267 platform::ArenaVector<const Column *> columns(arena);
268 columns.resize(node.output_attrs.size());
269 std::memset(columns.data(), 0, columns.size() * sizeof(const Column *));
270 auto *table = std::get<const ColumnarTable *>(input.data);
271
272 for (size_t i = 0; i < node.output_attrs.size(); ++i) {
273 auto [actual_col_idx, _] = node.output_attrs[i];
274 columns[i] = needed[i] ? &table->columns[actual_col_idx] : nullptr;
275 }
276 return columns;
277}

286inline void prepare_output_columns(
287 ColumnarReader &reader, const JoinInput &build_input,
288 const JoinInput &probe_input, const PlanNode &build_node,
289 const PlanNode &probe_node,
290 const std::vector<std::tuple<size_t, DataType>> &remapped_attrs,
291 size_t build_size) {
292
293 bool build_is_columnar = build_input.is_columnar();
294 bool probe_is_columnar = probe_input.is_columnar();
295
296 if (!build_is_columnar && !probe_is_columnar)
297 return;
298
299 auto &arena = Contest::platform::get_arena(0);
300
301 platform::ArenaVector<uint8_t> build_needed(arena);
302 build_needed.resize(build_node.output_attrs.size());
303 std::memset(build_needed.data(), 0, build_needed.size());
304
305 platform::ArenaVector<uint8_t> probe_needed(arena);
306 probe_needed.resize(probe_node.output_attrs.size());
307 std::memset(probe_needed.data(), 0, probe_needed.size());
308
309 for (const auto &[col_idx, dtype] : remapped_attrs) {
310 if (col_idx < build_size) {
311 if (build_is_columnar) {
312 build_needed[col_idx] = 1;
313 }
314 } else if (probe_is_columnar) {
315 probe_needed[col_idx - build_size] = 1;
316 }
317 }
318
319 if (build_is_columnar) {
320 reader.prepare_build(collect_needed_columns(build_input, build_node,
321 build_needed, arena));
322 }
323
324 if (probe_is_columnar) {
325 reader.prepare_probe(collect_needed_columns(probe_input, probe_node,
326 probe_needed, arena));
327 }
328}
