213 void save_page() {
214 auto *page = get_page();
215 *reinterpret_cast<uint16_t *>(page) = num_rows;
216 *reinterpret_cast<uint16_t *>(page + 2) =
217 static_cast<uint16_t>((data_end - data_begin()) / sizeof(T));
218 size_t bitmap_size = (num_rows + 7) / 8;
219 memcpy(page + PAGE_SIZE - bitmap_size, bitmap.data(), bitmap_size);
220 ++last_page_idx;
221 num_rows = 0;
222 data_end = data_begin();
223 }

228 bitmap[byte_idx] |= (0x1 << bit_idx);
229 }

231 void unset_bitmap(size_t idx) {

233 size_t bit_idx = idx % 8;
234 bitmap[byte_idx] &= ~(0x1 << bit_idx);
235 }

237 void insert(T value) {
238 if (data_end + 4 + num_rows / 8 + 1 > PAGE_SIZE) [[unlikely]] {
239 save_page();
240 }

243 data_end += sizeof(T);
244 set_bitmap(num_rows);

297 *reinterpret_cast<uint16_t *>(page + 2) = page_data_len;
