46template <MatchCollectionMode Mode>
47inline void eq_scan_build(uint32_t probe_id, int32_t probe_val,
48 const int32_t *build_vals, const uint32_t *build_ids,
49 size_t build_count,
50 ThreadLocalMatchBuffer<Mode> &buffer) {
51#if defined(__x86_64__) && defined(__AVX2__)
52 __m256i build_reg =
53 _mm256_load_si256(reinterpret_cast<const __m256i *>(build_vals));
54 const int valid_mask = (1 << build_count) - 1;
55
56 __m256i probe_reg = _mm256_set1_epi32(probe_val);
57 __m256i cmp = _mm256_cmpeq_epi32(probe_reg, build_reg);
58 int mask = _mm256_movemask_ps(_mm256_castsi256_ps(cmp)) & valid_mask;
59
60 while (mask) {
61 int idx = __builtin_ctz(mask);
62 buffer.add_match(build_ids[idx], probe_id);
63 mask &= mask - 1;
64 }
65
66#elif defined(__aarch64__)
67 int32x4_t build_lo = vld1q_s32(build_vals);
68 int32x4_t build_hi = vld1q_s32(build_vals + 4);
69 const uint32_t valid_mask = (1u << build_count) - 1;
70
71 static constexpr uint32_t mask_bits_lo[4] = {1, 2, 4, 8};
72 static constexpr uint32_t mask_bits_hi[4] = {16, 32, 64, 128};
73 const uint32x4_t bit_mask_lo = vld1q_u32(mask_bits_lo);
74 const uint32x4_t bit_mask_hi = vld1q_u32(mask_bits_hi);
75
76 int32x4_t probe_reg = vdupq_n_s32(probe_val);
77 uint32x4_t cmp_lo = vceqq_s32(probe_reg, build_lo);
78 uint32x4_t cmp_hi = vceqq_s32(probe_reg, build_hi);
79 uint32_t mask = vaddvq_u32(vandq_u32(cmp_lo, bit_mask_lo)) |
80 vaddvq_u32(vandq_u32(cmp_hi, bit_mask_hi));
81 mask &= valid_mask;
82
83 while (mask) {
84 int idx = __builtin_ctz(mask);
85 buffer.add_match(build_ids[idx], probe_id);
86 mask &= mask - 1;
87 }
88
89#else
90 for (size_t k = 0; k < build_count; ++k) {
91 if (build_vals[k] == probe_val) {
92 buffer.add_match(build_ids[k], probe_id);
93 }
94 }
95#endif
96}
